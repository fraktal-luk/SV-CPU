; Generate INV with enabled trap

@section prog_main

$start
	add_i r20, r0, 0x4000 ; initialize exc stack

	lds   r11, r0, 8
	add_i r11, r11, 0x07c0 ; set all 1's in FP trap enables
	sts   r11, r0, 8  ; enable FP exc (new)

	sys_sync

	addi_f f5, f2, f3
	addi_f f7, f2, f3
	
	inv_f  f11, f5, f5   ; generate exc

	jl     r31, $save_sync_exc

	add_i 	r1, r0, 2  ; OV
	jl 		r31, $get_cr8_bit
	jnz_i   r1, $error

	add_i 	r1, r0, 4  ; INV
	jl 		r31, $get_cr8_bit
	jz_i    r1, $error


$check
	sys_dbcall
$error
    sys_error

@include "common_code/exc_state.txt"
