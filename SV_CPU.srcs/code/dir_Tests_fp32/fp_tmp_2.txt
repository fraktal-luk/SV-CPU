; Generate INV and OV with disabled trap

@section prog_main

$start

	; FP op neutral
	addi_f f5, f2, f3
	
	; check fpStatus
	sys_sync

	add_i 	r1, r0, 4  ; INV
	jl 		r31, $get_cr8_bit
	jnz_i    r1, $error

	add_i 	r1, r0, 2  ; OV
	jl 		r31, $get_cr8_bit
	jnz_i    r1, $error

	; FP op, neutral
	addi_f f7, f2, f3

	
	; Set 2 exc bits in short sequence
	inv_f  f11, f5, f5
	ov_f   f22, f21, f21

	; Simple ops, neutral
	addi_f f5, f2, f3
	addi_f f5, f2, f3


	; check fpStatus == {INV: 1, OV: 1, default: 0}
	sys_sync

	add_i 	r1, r0, 4  ; INV
	jl 		r31, $get_cr8_bit
	jz_i    r1, $error

	add_i 	r1, r0, 2  ; OV
	jl 		r31, $get_cr8_bit
	jz_i    r1, $error


$check
	sys_dbcall
$error
    sys_error
