; Generate INV with disabled trap

@section prog_main

$start

	addi_f f5, f2, f3

	sys_sync

	add_i 	r1, r0, 4  ; INV
	jl 		r31, $get_cr8_bit
	jnz_i   r1, $error

	add_i 	r1, r0, 2  ; OV
	jl 		r31, $get_cr8_bit
	jnz_i   r1, $error


	; generate INV
	inv_f  f11, f5, f5
	
	; check fpStatus == {INV: 1, default: 0}
	sys_sync

	add_i 	r1, r0, 4  ; INV
	jl 		r31, $get_cr8_bit
	jz_i   r1, $error

	add_i 	r1, r0, 2  ; OV
	jl 		r31, $get_cr8_bit
	jnz_i   r1, $error


	; FP op, normal
	addi_f f20, f4, f2

	; Gen OV
	ov_f   f22, f21, f21

	; check fpStatus == {INV: 1, OV: 1, default: 0}

	sys_sync

	add_i 	r1, r0, 4  ; INV
	jl 		r31, $get_cr8_bit
	jz_i   r1, $error

	add_i 	r1, r0, 2  ; OV
	jl 		r31, $get_cr8_bit
	jz_i   r1, $error


$check
	sys_dbcall
$error
    sys_error
