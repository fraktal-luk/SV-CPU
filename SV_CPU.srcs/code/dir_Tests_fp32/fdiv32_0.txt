
@section prog_main
	
	jl r31, $get_adr_data_main
	or_r  r8, r0, r1

	jl r31, $get_adr_data_output
	or_r  r9, r0, r1


	add_i 	r1, r0, 1
	jl 		r31, $set_rm  ; rm = 1 / RM_Up

	add_i r10, r0, 3

$loop
	ldf_i  f2, r8, 0
	ldf_i  f3, r8, 4

	divf32 f10, f2, f3

	stf_i  f10, r9, 0

	add_i  r8, r8, 8
	add_i  r9, r9, 4

	add_i r10, r10, -1
	jnz_i r10, $loop


	; check Inexact (from min subnormal by 2)
	sys_sync
	add_i 	r1, r0, 0  ; INEXACT
	jl 		r31, $get_cr8_bit
	jnz_i 	r1, $error

	sys_dbcall
	sys_error
