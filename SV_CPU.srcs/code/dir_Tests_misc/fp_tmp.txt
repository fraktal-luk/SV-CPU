;
@section prog_main

$start
	add_i r3, r0, 1
	shl_i r3, r3, 15  ; fp en bit

	add_i r1, r0, 1
	shl_i r1, r1, 31  ; INV set
	or_r r1, r1, r3

	add_i r2, r0, 3
	shl_i r2, r2, 30  ; INV | OV set
	or_r  r2, r2, r3

	addi_f f5, f2, f3
	
	; check fpStatus == 0
	lds    r10, r0, 8
	sub_r  r10, r10, r3
	jnz_i  r10, $error
	
	inv_f  f11, f5, f5
	
	; check fpStatus == {INV: 1, default: 0}
	sys_sync
	lds   r10, r0, 8
	sub_r r11, r10, r1
	jnz_i r11, $error	
	
	addi_f f20, f4, f2
	ov_f   f22, f21, f21

	; check fpStatus == {INV: 1, OV: 1, default: 0}
	sys_sync
	lds   r10, r0, 8
	sub_r r11, r10, r2
	jnz_i r11, $error

$check
	sys_dbcall
$error
    sys_error
